cmake_minimum_required(VERSION 3.16)
project(edge_frame_host_tests C CXX)
enable_testing()
add_executable(test_frame test_frame.c ../main/frame.c)
target_include_directories(test_frame PRIVATE ../main)
target_compile_features(test_frame PRIVATE c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_frame PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME firmware_frame COMMAND test_frame)
add_executable(test_compute_worker test_compute_worker.c)
target_include_directories(test_compute_worker PRIVATE ../compute-worker/src)
target_compile_features(test_compute_worker PRIVATE c_std_11)
target_link_libraries(test_compute_worker PRIVATE m)
add_executable(test_worker_runtime test_worker_runtime.cpp)
target_include_directories(test_worker_runtime PRIVATE arduino_stub ../compute-worker/src)
target_compile_features(test_worker_runtime PRIVATE cxx_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_compute_worker PRIVATE -Wall -Wextra -Werror -pedantic)
    target_compile_options(test_worker_runtime PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME compute_worker COMMAND test_compute_worker)
add_test(NAME worker_runtime COMMAND test_worker_runtime)
